**Sơ đồ hoạt động (Activity Diagram) - UC17: Đăng ký nội trú**

```plantuml
@startuml
|Sinh viên|
start
:Truy cập chức năng đăng ký;
|Hệ thống|
:Kiểm tra trạng thái nội trú hiện tại;
if (Đã có phòng đang hoạt động?) then (Có)
  :Hiển thị "Bạn đã có phòng, không thể đăng ký mới";
  :Ghi log kiểm tra trạng thái;
  stop
else (Không)
endif

repeat
  :Hiển thị form đăng ký;
  |Sinh viên|
  :Nhập thông tin (thời gian, loại phòng, ghi chú...);
  |Hệ thống|
  :Validate client;
  
  if (Thiếu thông tin bắt buộc?) then (Có)
    :Highlight các trường thiếu;
    :Hiển thị "Vui lòng nhập đầy đủ thông tin!";
  elseif (Dữ liệu sai định dạng?) then (Có)
    :Chặn submit, hiển thị lỗi chi tiết từng trường;
    :Giữ dữ liệu hợp lệ, yêu cầu sửa dữ liệu sai;
  else (Hợp lệ)
    |Sinh viên|
    :Nhấn "Gửi đăng ký";
    |Hệ thống|
    :Gửi request lên server;
    :Server validate dữ liệu;
    
    if (Đã tồn tại request trạng thái Pending?) then (Có)
      :Từ chối tạo mới, Trả response lỗi;
      :Hiển thị "Bạn đã có yêu cầu đang chờ xử lý!";
      :Ghi log business rule;
      stop
    else (Chưa có)
      :Tạo bản ghi (Pending);
      :Lưu DB;
      if (Lỗi DB?) then (Có)
        :Rollback transaction;
        :Trả response lỗi hệ thống\nHiển thị "Có lỗi xảy ra, vui lòng thử lại sau";
        :Ghi exception log;
        stop
      else (Thành công)
        :Ghi log;
        :Trả response thành công\nHiển thị thông báo thành công;
        stop
      endif
    endif
  endif
repeat while (Lỗi nhập liệu?) is (Có)
@enduml
```

**Sơ đồ hoạt động (Activity Diagram) - UC08: Xét duyệt đăng ký**

```plantuml
@startuml
|Nhân viên|
start
:Mở danh sách đăng ký;
|Hệ thống|
:Tải danh sách Pending;
if (Không có dữ liệu?) then (Có)
  :Hiển thị "Không có dữ liệu";
  stop
else (Có dữ liệu)
endif

repeat
  |Nhân viên|
  :Chọn 1 request;
  |Hệ thống|
  if (Chưa chọn yêu cầu?) then (Có)
    :Chặn hành động bước tiếp theo;
    :Hiển thị "Vui lòng chọn yêu cầu";
    backward: Quay lại bước chọn;
  else (Đã chọn)
    :Xem chi tiết;
    |Nhân viên|
    :Chọn Duyệt/Từ chối;
    :Gửi request;
    |Hệ thống|
    :Server kiểm tra trạng thái;
    
    if (Không tìm thấy request trong DB?) then (Có)
      :Trả response lỗi, Hiển thị "Yêu cầu không tồn tại";
      :Ghi log;
      stop
    elseif (Request không còn Pending?) then (Có)
      :Từ chối xử lý, không cập nhật DB;
      :Hiển thị "Yêu cầu đã được xử lý", Ghi log;
      :Refresh danh sách;
      backward: Quay lại bước chọn;
    else (Hợp lệ)
      :Cập nhật trạng thái;
      :Lưu DB;
      if (Lỗi DB?) then (Có)
        :Rollback transaction;
        :Trả lỗi hệ thống, Hiển thị lỗi, Ghi exception log;
        stop
      else (Thành công)
        :Ghi log;
        :Trả response thành công, Hiển thị thông báo thành công;
        stop
      endif
    endif
  endif
repeat while (Xử lý request tiếp theo?)
@enduml
```

        :Trả lỗi hệ thống, Hiển thị lỗi, Ghi exception log;

      else (Thành công)
        :Ghi log, Trả response thành công;
        :Hiển thị thành công;

      endif
    endif
  endif

@enduml
